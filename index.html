// MXU_fullSync.js import { processTXT } from "./txt/raw.txt"; import { AXINXA } from "./AXINXA.js"; import { NC } from "./NC/index.js"; import { MXU_TREUE } from "./MXU_TREUE.js"; export function MXU_fullSync(packet = {}) { const time = Date.now(); const msg = String(packet.msg || ""); // PX const px = processTXT({ msg }); // META (RESPO entfernt) const meta = { origin: "MXU", msg, pxType: typeof px, time }; // AXINXA const axis = { AX: AXINXA.runAll(), NEW: AXINXA.runNew() }; // NC const nc = NC(px, meta); // Treue-Bewertung const treue = MXU_TREUE.accept({ input: msg, px, meta, axis, nc }); return { sync: true, stable: treue.status === "OK", time, input: msg, px, meta, axis, nc, treue, info: "MXU_fullSync → Synchronisation + Systemtreue aktiv" }; }
