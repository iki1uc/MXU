"45°" : "0°" }; }, // XA wird in NC.SENSOR sichtbar sensor() { return { ok: true, angle: XA.angle, maneuver: XA.maneuver }; }, // XA wird in NC.REPORT sichtbar report() { return { type: "XA-ATOM", angle: XA.angle, maneuver: XA.maneuver, mode: XA.maneuver ? "MANEUVER" : "LINEAR" }; } }, // ─────────────────────────────────────────────── // ◈ MXU-HOOKS (Atom-Modus, keine Masse) // ─────────────────────────────────────────────── mxu: { attach(MXU) { MXU.XA = XA; return true; } }, // ─────────────────────────────────────────────── // ◈ SYSTEM-SENSOR (lokal) // ─────────────────────────────────────────────── sensor() { return { angle: this.angle, maneuver: this.maneuver, atom: true }; }, // ─────────────────────────────────────────────── // ◈ SYSTEM-ARBEITER (lokal) // ─────────────────────────────────────────────── worker(cmd) { switch(cmd) { case "0°": this.angle = 0; this.maneuver = false; break; case "45°": this.round(); break; } return this.sensor(); }, // ─────────────────────────────────────────────── // ◈ MINI-STAGE (XA-eigene Stage) // ─────────────────────────────────────────────── stage: { 0: () => XA.worker("0°"), 1: () => XA.worker("45°") } };
